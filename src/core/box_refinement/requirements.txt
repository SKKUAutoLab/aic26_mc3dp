# box_refinement — Python dependencies
#
# Pinned to the exact versions the refinement was validated on. The project's own pyproject.toml
# declares no dependencies, so this file is self-contained: it is everything needed to run the two
# refine_scene* entry points on a fresh machine, and nothing else.
#
#   conda create -n refinement python=3.12 -y && conda activate refinement
#   pip install -r src/matching/core/box_refinement/requirements.txt
#
# TORCH IS A CUDA BUILD and does not come from plain PyPI. Install it first, matching the machine's
# driver (this was validated on CUDA 13.0, 2x RTX A6000):
#
#   pip install torch==2.12.0 torchvision==0.27.0 --index-url https://download.pytorch.org/whl/cu130
#
# then the rest of this file. Check `nvidia-smi` and pick the cu### that matches; the refinement
# does not care which CUDA it is, only that torch sees the GPU.
